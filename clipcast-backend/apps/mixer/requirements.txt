# clipcast-mixer — Audio Studio engine (generate + mashup)
# This app is fully independent of the processor/downloader images; it has its
# own torch pin so a music-model bump can never disturb WhisperX's pins.

# ── PyTorch (cu121) ────────────────────────────────────────────────────────────
torch==2.4.1
torchaudio==2.4.1

# ── Stem separation ────────────────────────────────────────────────────────────
# Demucs v4 (htdemucs). MIT-licensed, state-of-the-art open separation.
demucs>=4.0.1

# ── Music analysis (BPM / beat grid / key) ────────────────────────────────────
# librosa covers beat tracking + chroma for key detection with no native-build
# pain. madmom (better downbeats) and essentia (better key) are documented
# upgrades in docs/15 — swap in once the happy path is proven on a deploy.
librosa>=0.10.1
numpy>=1.26.0,<2.0
scipy>=1.11.0
soundfile>=0.12.1

# ── Tempo-stretch / pitch-shift (high quality) ────────────────────────────────
# pyrubberband shells out to the `rubberband` CLI (apt-installed in the image).
pyrubberband>=0.3.0

# ── Loudness normalization (EBU R128) ─────────────────────────────────────────
pyloudnorm>=0.1.1

# ── Reference mastering (make the mix sound "released") ───────────────────────
# Matchering matches the final mix's EQ / RMS / peak / stereo-width to a
# professionally-mastered reference — we use each source song's own original
# recording as the reference. GPL-3, server-side only; reference_master() falls
# back to pyloudnorm if it can't run, so the master never fails.
matchering>=2.0.5

# ── Audio quality rating (closes the feedback loop) ───────────────────────────
# Meta Audiobox-Aesthetics predicts Production Quality / Complexity / Content
# Enjoyment / Usefulness (1-10). The mashup re-renders and keeps the highest
# Production-Quality take. MIT/CC — see docs/16.
audiobox_aesthetics>=0.0.1

# NOTE: faster-whisper, basic-pitch, pretty_midi, pyfluidsynth and langfuse are
# installed in a SEPARATE image layer (see main.py) — bundling them here made
# pip's resolver explode ("resolution-too-deep") against the pinned ML core.

# ── AI arrangement brain (which sections to mix, DJ-style) ────────────────────
# Reuses the same Gemini the clip pipeline already uses (docs/13); falls back to
# a deterministic energy-based arrangement when unavailable.
google-genai>=2.0.0

# ── BandLab-style effects chain (EQ / comp / reverb / limiter) ────────────────
# Spotify's pedalboard gives VST-quality processing used to finish each hook part
# and the transitions. GPL-3, server-side only. apply_fx_chain() falls back to a
# scipy high-pass + ceiling if it can't load, so FX never breaks the render.
pedalboard>=0.9.0

# ── Text→music generation ─────────────────────────────────────────────────────
# transformers MusicGen is the codepath here because it is the most reliable to
# deploy unattended. To move to the commercially-safe ACE-Step (Apache-2.0, see
# docs/14), swap `_generate_music()` — the rest of the pipeline is unchanged.
# Cap below 4.49: transformers 4.49+ added the CVE-2025-32434 mitigation that
# makes torch.load refuse to run on torch < 2.6 for non-safetensors checkpoints
# (facebook/musicgen-medium ships .bin weights), which 502s the generate worker.
# We're pinned to torch 2.4.1 here, so stay on the pre-gate line. MusicGen is
# stable on 4.44-4.48. (Alternative: bump torch>=2.6 in lockstep — heavier.)
transformers>=4.44.0,<4.49
accelerate>=0.33.0

# (Melody re-instrumentation deps — basic-pitch/pretty_midi/pyfluidsynth — are
# installed in a separate layer in main.py, see the note above.)

# ── AWS / S3 ───────────────────────────────────────────────────────────────────
boto3>=1.34.0

# ── Web framework / Modal ─────────────────────────────────────────────────────
fastapi[standard]>=0.110.0
modal>=0.64.0
